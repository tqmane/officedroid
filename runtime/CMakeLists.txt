cmake_minimum_required(VERSION 3.22)
project(officedroid_runtime C)
add_executable(odprobe probe.c)
target_compile_options(odprobe PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    set_target_properties(odprobe PROPERTIES OUTPUT_NAME libodprobe.so)
    target_link_options(odprobe PRIVATE -Wl,-z,max-page-size=16384)
else()
    enable_testing()
    add_test(NAME process_filesystem_memory COMMAND odprobe ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(process_filesystem_memory PROPERTIES PASS_REGULAR_EXPRESSION "\"ok\":true")
    add_test(NAME invalid_directory COMMAND odprobe /officedroid-nonexistent-directory)
    set_tests_properties(invalid_directory PROPERTIES WILL_FAIL TRUE)
endif()
